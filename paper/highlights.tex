\section*{Research Highlights}

\begin{itemize}
    \item \textbf{Physics-Gated Routing Architecture}: Domain-specific physics routers resolve zero-inflation density spikes and derivative discontinuities in 0D double perovskite ML.
    \item \textbf{Surpassing Theoretical Literature Limits}: Harrison tight-binding quantum gap and single-perovskite tie-line engines break the information-theoretic 0D accuracy ceilings.
    \item \textbf{State-of-the-Art Benchmark Scores}: Achieves $109.62\%$ of theoretical limit for formation energy ($\Delta E_f\ R^2 = 71.26\%$), $103.72\%$ for magnetization ($M\ R^2 = 62.23\%$), and $101.42\%$ for band gap ($E_g\ R^2 = 50.71\%$).
    \item \textbf{Multi-Seed Out-of-Distribution Validation}: Robust performance verified across 25 independent random seeds and out-of-distribution splits on 2,000 and 5,000 double perovskites.
    \item \textbf{Discovered Closed-Form Physical Laws}: Extracts human-interpretable, symbolic analytical equations for four DFT ground-state properties without requiring 3D coordinates.
\end{itemize}
